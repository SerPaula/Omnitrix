    <script>
        let alienData = [];

        // Listener para la carga manual del archivo
        document.getElementById('csvFile').addEventListener('change', function(e) {
            const file = e.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(event) {
                parseCSV(event.target.result);
                document.getElementById('uploadSection').classList.add('hidden');
                document.getElementById('searchSection').classList.remove('hidden');
            };
            reader.readAsText(file);
        });

        function parseCSV(csvText) {
            const lines = csvText.split(/\r?\n/);
            const headers = lines[0].split(',').map(h => h.trim());
            
            alienData = [];
            for (let i = 1; i < lines.length; i++) {
                if (!lines[i].trim()) continue;
                
                // Regex para manejar comas dentro de comillas (común en la columna de poderes)
                const values = lines[i].match(/(".*?"|[^",]+)/g).map(v => v.replace(/^"|"$/g, '').trim());
                
                let obj = {};
                headers.forEach((header, index) => {
                    obj[header] = values[index] || "";
                });
                alienData.push(obj);
            }
            
            document.getElementById('statusMessage').textContent = `> ACCESO CONCEDIDO: ${alienData.length} ADN IDENTIFICADOS`;
            displayAliens(alienData.slice(0, 4)); // Mostrar algunos de ejemplo
        }

        function displayAliens(data) {
            const grid = document.getElementById('resultsGrid');
            grid.innerHTML = '';

            if (data.length === 0) {
                grid.innerHTML = `
                    <div class="col-span-full text-center py-20 border border-red-900 bg-red-900/10 rounded-lg">
                        <p class="text-red-500 font-bold uppercase tracking-tighter">Fallo en el Escaneo: ADN No Encontrado</p>
                    </div>
                `;
                return;
            }

            data.forEach(alien => {
                const card = document.createElement('div');
                card.className = 'card p-6 rounded-lg border border-zinc-800 flex flex-col';
                
                card.innerHTML = `
                    <div class="flex justify-between items-start mb-4">
                        <div>
                            <h2 class="text-2xl font-black text-green-400 glow-text uppercase">${alien.name}</h2>
                            <span class="text-[10px] bg-green-900 text-green-200 px-2 py-0.5 rounded font-bold">${alien.series}</span>
                        </div>
                        <div class="text-right">
                            <span class="text-zinc-600 text-[10px] block font-bold uppercase">Planeta</span>
                            <span class="text-xs text-white">${alien.home_world}</span>
                        </div>
                    </div>
                    
                    <div class="mb-6 flex-1">
                        <div class="bg-black/50 p-3 rounded border-l-2 border-green-500">
                            <span class="text-green-800 text-[10px] block font-black uppercase mb-1">Capacidades de Combate</span>
                            <p class="text-sm text-zinc-300 leading-tight italic">${alien.power || 'N/A'}</p>
                        </div>
                    </div>

                    <div class="grid grid-cols-3 gap-2 pt-4 border-t border-zinc-800">
                        <div class="text-center">
                            <span class="text-[9px] text-zinc-500 block font-bold uppercase">Fuerza</span>
                            <div class="w-full bg-zinc-800 h-1 mt-1 rounded-full overflow-hidden">
                                <div class="bg-green-500 h-full" style="width: ${Math.min(alien.combat || 0, 100)}%"></div>
                            </div>
                            <span class="text-xs font-bold text-white mt-1 block">${alien.combat || 0}</span>
                        </div>
                        <div class="text-center">
                            <span class="text-[9px] text-zinc-500 block font-bold uppercase">Velocidad</span>
                             <div class="w-full bg-zinc-800 h-1 mt-1 rounded-full overflow-hidden">
                                <div class="bg-blue-500 h-full" style="width: ${Math.min(alien.speed || 0, 100)}%"></div>
                            </div>
                            <span class="text-xs font-bold text-white mt-1 block">${alien.speed || 0}</span>
                        </div>
                        <div class="text-center">
                            <span class="text-[9px] text-zinc-500 block font-bold uppercase">Inteligencia</span>
                             <div class="w-full bg-zinc-800 h-1 mt-1 rounded-full overflow-hidden">
                                <div class="bg-purple-500 h-full" style="width: ${Math.min(alien.intelligence || 0, 100)}%"></div>
                            </div>
                            <span class="text-xs font-bold text-white mt-1 block">${alien.intelligence || 0}</span>
                        </div>
                    </div>
                `;
                grid.appendChild(card);
            });
        }

        function filterAliens() {
            const query = document.getElementById('alienSearch').value.toLowerCase();
            const loader = document.getElementById('loader');
            const grid = document.getElementById('resultsGrid');

            if (!query) {
                displayAliens(alienData.slice(0, 4));
                return;
            }

            loader.classList.remove('hidden');
            grid.classList.add('opacity-0');

            setTimeout(() => {
                const filtered = alienData.filter(alien => 
                    alien.name.toLowerCase().includes(query) || 
                    alien.species.toLowerCase().includes(query) ||
                    alien.power.toLowerCase().includes(query)
                );
                
                displayAliens(filtered);
                loader.classList.add('hidden');
                grid.classList.remove('opacity-0');
                grid.classList.add('animate-in', 'fade-in', 'duration-500');
            }, 400);
        }

        document.getElementById('alienSearch').addEventListener('keypress', function (e) {
            if (e.key === 'Enter') filterAliens();
        });
    </script>
